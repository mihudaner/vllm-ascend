cmake_minimum_required(VERSION 3.20)
project(csa_aclnn_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CSA_VENDOR "" CACHE PATH "Installed CSA OPP vendor directory")
if(NOT EXISTS "${CSA_VENDOR}/op_api/include/aclnn_csa_spmd_tnd_pto.h")
  message(FATAL_ERROR "Set CSA_VENDOR to the installed custom vendor")
endif()
add_executable(csa_aclnn_test main.cpp)
target_include_directories(csa_aclnn_test PRIVATE "$ENV{ASCEND_HOME_PATH}/include" "${CSA_VENDOR}/op_api/include")
target_link_directories(csa_aclnn_test PRIVATE "$ENV{ASCEND_HOME_PATH}/lib64" "${CSA_VENDOR}/op_api/lib")
target_link_libraries(csa_aclnn_test PRIVATE ascendcl cust_opapi nnopbase dl)
set_target_properties(csa_aclnn_test PROPERTIES BUILD_RPATH "${CSA_VENDOR}/op_api/lib;$ENV{ASCEND_HOME_PATH}/lib64")
